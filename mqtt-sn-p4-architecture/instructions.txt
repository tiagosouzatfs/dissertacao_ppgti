### Entrar na pasta do projeto ###

cd ~/dissertacao_ppgti


-----------------------------------------------------------------------------


### Compilação do programa p4###


docker run -dit --name=p4c --rm ramonfontes/bmv2:latest

docker cp mqtt-sn-p4-architecture/p4src/mqtt-sn.p4 p4c:/tmp/

docker exec -it p4c bash

cd /tmp

p4c --target bmv2 --arch v1model mqtt-sn.p4

exit

docker cp p4c:/tmp/mqtt-sn.json mqtt-sn-p4-architecture/


------------------------------------------------------------------------------
