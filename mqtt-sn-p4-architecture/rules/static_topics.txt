# Caso o topicName não esteja registrado, retorna REGACK com código de rejeição
table_set_default topic_registry send_regack_response_reject_invalid

# Exemplo: topicName 0x1234 é registrado como topicId 10
table_add topic_registry send_regack_response_accept 0x1234 => 10

# Exemplo: topicName 0x5678 é registrado como topicId 20
table_add topic_registry send_regack_response_accept 0x5678 => 20

# Exemplo: topicName 0x9ABC é registrado como topicId 30
table_add topic_registry send_regack_response_accept 0x9ABC => 30
