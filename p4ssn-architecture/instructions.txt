### Entrar na pasta do projeto ###

cd ~/dissertacao_ppgti


-----------------------------------------------------------------------------


### Compilação do programa p4###


docker run -dit --name=p4c --rm ramonfontes/bmv2:latest

docker cp p4ssn-architecture/p4src/p4ssn.p4 p4c:/tmp/

docker exec -it p4c bash

cd /tmp

p4c --target bmv2 --arch v1model p4ssn.p4

exit

docker cp p4c:/tmp/p4ssn.json p4ssn-architecture/
