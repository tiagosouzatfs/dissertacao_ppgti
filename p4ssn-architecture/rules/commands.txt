# --- GRUPO MULTICAST PARA BROADCAST (8 portas) ---
# Como os nomes das tebelas e actions são iguais no ingress e egresss tem que passar o nome todo

# --- GRUPO MULTICAST ACELERADO PARA TÓPICO 20 (5 Clientes Sub) ---
# Como os nomes das tebelas e actions são iguais no ingress e egresss tem que passar o nome todo

# Tópico 20 (também funciona com shorttopicname e topicname) -> Grupo Multicast 20
table_add MyIngress.multicast_acelerate_forwarding MyIngress.multicast_acelerate 20 => 20

# Configuração do Egress para reescrever MAC e IP por porta para Multicast de vários clientes sub

table_add MyEgress.multicast_acelerate_forwarding MyEgress.multicast_acelerate 5 => 00:00:00:00:00:05 10.0.0.5 1895

table_add MyEgress.multicast_acelerate_forwarding MyEgress.multicast_acelerate 7 => 00:00:00:00:00:07 10.0.0.7 1897
table_add MyEgress.multicast_acelerate_forwarding MyEgress.multicast_acelerate 8 => 00:00:00:00:00:08 10.0.0.8 1898


# --- UNICAST ACELERADO PARA OUTROS TÓPICOS ---

# Tópico 10 -> Unicast
table_add MyIngress.unicast_acelerate_forwarding MyIngress.unicast_acelerate 10 => 00:00:00:00:00:04 4 10.0.0.4 1894
# Tópico 0x6162 é representação hexadecimal do shorttopicname Unicast "ab"
